cmake_minimum_required(VERSION 3.20)
project(kenai_tray_native_test LANGUAGES CXX RC)
set(RUNNER "${CMAKE_CURRENT_SOURCE_DIR}/../../windows/runner")
add_executable(kenai_tray_test desktop_tray_test.cpp "${RUNNER}/desktop_tray.cpp" "${RUNNER}/Runner.rc")
target_compile_features(kenai_tray_test PRIVATE cxx_std_17)
target_compile_options(kenai_tray_test PRIVATE /W4 /WX /utf-8)
target_compile_definitions(kenai_tray_test PRIVATE UNICODE _UNICODE)
target_include_directories(kenai_tray_test PRIVATE "${RUNNER}")
target_link_libraries(kenai_tray_test PRIVATE shell32 user32)
